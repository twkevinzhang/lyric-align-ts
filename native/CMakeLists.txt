cmake_minimum_required(VERSION 3.16)
project(whisper_parity LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
option(POC_WASM_ARCH "Select GGML's explicit wasm quantized kernels" OFF)
set(POC_PTHREAD_POOL_SIZE 1 CACHE STRING "Prewarmed pthread worker pool")
if(EMSCRIPTEN AND POC_WASM_ARCH)
  # Emscripten 4.0.22 reports x86; this fork checks the processor for wasm.
  set(CMAKE_SYSTEM_PROCESSOR wasm32)
endif()
add_subdirectory(${WHISPER_SOURCE} whisper)
target_include_directories(whisper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(parity bridge.cpp runner.cpp)
target_link_libraries(parity PRIVATE whisper)
if(POC_TIME_SCALE)
  target_compile_definitions(parity PRIVATE POC_TIME_SCALE=1)
endif()
if(POC_UNROLL_TEST)
  add_executable(unroll_test unroll-test.cpp)
  target_link_libraries(unroll_test PRIVATE ggml-cpu ggml-base)
  target_include_directories(unroll_test PRIVATE ${WHISPER_SOURCE}/ggml/src ${WHISPER_SOURCE}/ggml/src/ggml-cpu)
  target_compile_options(unroll_test PRIVATE -pthread -msimd128)
  if(POC_WEIGHT_CACHE)
    target_compile_definitions(unroll_test PRIVATE POC_WEIGHT_CACHE_TEST=1)
  endif()
  if(POC_FLOAT_REORDER)
    target_compile_definitions(unroll_test PRIVATE POC_FLOAT_REORDER=1)
  endif()
  if(POC_QUANT_FMA)
    target_compile_definitions(unroll_test PRIVATE POC_QUANT_FMA=1)
  endif()
  if(POC_QUANT_DOT)
    target_compile_definitions(unroll_test PRIVATE POC_QUANT_DOT=1)
  endif()
  if(POC_QUANT_REORDER)
    target_compile_definitions(unroll_test PRIVATE POC_QUANT_REORDER=1)
  endif()
  set(POC_TEST_POOL ${POC_PTHREAD_POOL_SIZE})
  if(POC_TEST_POOL LESS 6)
    set(POC_TEST_POOL 6)
  endif()
  target_compile_definitions(unroll_test PRIVATE POC_TEST_MAX_THREADS=${POC_TEST_POOL})
  target_link_options(unroll_test PRIVATE -pthread "-sENVIRONMENT=node" "-sPTHREAD_POOL_SIZE=${POC_TEST_POOL}" "-sEXIT_RUNTIME=1")
  set_target_properties(unroll_test PROPERTIES SUFFIX ".cjs")
endif()
if(POC_TRACE)
  target_compile_definitions(parity PRIVATE POC_TRACE)
  target_include_directories(whisper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endif()
if(EMSCRIPTEN)
  target_compile_definitions(parity PRIVATE POC_MAX_THREADS=${POC_PTHREAD_POOL_SIZE})
  target_compile_options(parity PRIVATE -pthread -msimd128)
  target_link_options(parity PRIVATE
    -pthread "-sPTHREAD_POOL_SIZE=${POC_PTHREAD_POOL_SIZE}" "-sPTHREAD_POOL_SIZE_STRICT=2" "-sALLOW_BLOCKING_ON_MAIN_THREAD=0" "-sSTACK_SIZE=5242880"
    "-sMODULARIZE=1" "-sEXPORT_NAME=createWhisperParity" "-sENVIRONMENT=web,worker"
    "-sALLOW_MEMORY_GROWTH=1" "-sINITIAL_MEMORY=536870912" "-sMAXIMUM_MEMORY=2147483648"
    "-sFORCE_FILESYSTEM=1" "-sINVOKE_RUN=0"
    "-sEXPORTED_RUNTIME_METHODS=['FS','callMain','ccall','HEAPU8']"
    "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free','_poc_open','_poc_open_model','_poc_close','_poc_encode','_poc_encode_samples','_poc_decode','_poc_set_threads','_poc_get_threads','_poc_set_time_scale','_poc_set_audio_context','_poc_set_vocab_subset','_poc_mel_ms','_poc_encoder_ms','_poc_vocab','_poc_logits','_poc_qk_dims','_poc_qk_copy','_poc_trace_begin','_poc_trace_end']")
endif()
